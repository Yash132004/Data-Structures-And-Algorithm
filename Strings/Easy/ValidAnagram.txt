class Solution { // O(nlogn);
public:
    bool isAnagram(string s, string t) {
        int len1 = s.length(), len2 = t.length();
        if(len1 != len2){
            return false;
        }
        sort(s.begin(),s.end());
        sort(t.begin(), t.end());
        int i = 0, j=0;
        while( i < len1 && j < len2){
            if(s[i++] != t[j++]){
                return false;
            }
        }
        return true;
    }
};
