class Solution {
    #define ll long long
public:
    int calculate(string s) {
        stack<int> st;

        ll result = 0;
        ll num = 0;
        ll sign = 1;

        for (int i = 0; i < s.length(); i++) {

            if (isdigit(s[i])) {
                num = num * 10 + (s[i] - '0');
            }

            else if (s[i] == '+' || s[i] == '-') {

                result += sign * num;
                num = 0;

                if (s[i] == '+')
                    sign = 1;
                else
                    sign = -1;
            }

            else if (s[i] == '(') {

                st.push(result);
                st.push(sign);

                result = 0;
                sign = 1;
            }

            else if (s[i] == ')') {

                result += sign * num;
                num = 0;

                int prevSign = st.top();
                st.pop();

                int prevResult = st.top();
                st.pop();

                result = prevResult + prevSign * result;
                sign = 1;
            }
        }

        result += sign * num;

        return result;
    }
};
