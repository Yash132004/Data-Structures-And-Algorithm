class Solution {
public:
    int myAtoi(string s) {
        int i = 0;
        int n = s.size();
        int sign = 1;
        long num = 0;

        while(i<n && s[i] == ' ') i++; // skips the leading space

        if(i < n && (s[i] == '+' || s[i] == '-')){ // handling sign once only
            if(s[i] == '-') sign = -1;
            i++;
        }

        while(i < n && isdigit(s[i])){
            num = num*10 + (s[i]-'0');

            //checking overflow condition
            if(sign == 1 && num > INT_MAX) return INT_MAX;
            if(sign == -1 && -num < INT_MIN) return INT_MIN;

            i++;
        }
        return num*sign;
    }
};
