class Solution {
public:
    string multiply(string num1, string num2) {
        if(num1 == "0" || num2 == "0") return "0";

        int n = num1.size();
        int m = num2.size();

        vector<int> res(n+m, 0);

        for(int i = n-1; i>=0; i--){
            for(int j = m-1; j>=0; j--){
                int product = (num1[i] - '0')*(num2[j] - '0');
                int pos = i + j + 1; 

                res[pos] += product;
                res[pos-1] += res[pos] / 10;
                res[pos] %= 10;
            }
        }

        string ans = "";
        for(int digit: res){
            if(!(ans.empty() && digit==0)) // skiping leading zero's
                ans.push_back(digit + '0');
        }
        return ans;
    }
};
