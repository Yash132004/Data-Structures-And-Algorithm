class Solution {
public:
    string longestPalindrome(string s) {
        int maxi = 0;
        string palin = "";

        for (int i = 0; i < s.length(); i++) {
            for (int j = i; j < s.length(); j++) {

                int l = i, r = j;

                while (l <= r) {
                    if (s[l] != s[r])
                        break;
                    l++;
                    r--;
                }

                if (l > r && (j - i + 1) > maxi) {
                    maxi = j - i + 1;
                    palin = "";

                    for (int k = i; k <= j; k++)
                        palin += s[k];
                }
            }
        }

        return palin;
    }
};
