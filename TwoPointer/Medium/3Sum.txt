class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) { //TC: O(nlogn + n^2);
        int n = nums.size();
        vector<vector<int>>ans; //SC: O(unique Triplets)
        
        sort(nums.begin(),nums.end());

        for(int i =0; i<n ; i++){
            if(i>0 && nums[i] == nums[i-1]) continue; // avoid duplicate pair
            int j = i+1, k = n-1;

            while(j<k){
                int sum = nums[i] + nums[j] + nums[k];
                if(sum < 0){
                    j++;
                }else if(sum > 0){
                    k--;
                }else{ // sum == 0
                    ans.push_back({nums[i],nums[j],nums[k]});
                    j++; k--;

                    while(j < k && nums[j] == nums[j-1]) j++; // to avoid duplicate pair
                }
            }
        }
        
        return ans;
    }
};
