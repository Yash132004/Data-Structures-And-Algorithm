class Solution {
public:
    int trap(vector<int>& height) {
        int n = height.size();
        vector<int>rmax(n,-1);
        vector<int>lmax(n,-1);
        int maxleft = height[0];
        int maxright = height[n-1];
        for(int i = 1; i < n; i++){
            maxleft = max(maxleft, height[i-1]);
            lmax[i] = maxleft;
        }

        for(int i = n-2; i >= 0; i--){
            maxright = max(maxright, height[i+1]);
            rmax[i] = maxright;
        }
        int water = 0;
        for(int i = 0; i < n; i++){
            if(lmax[i] != -1 && rmax[i] != -1){
                int mini = min(lmax[i], rmax[i]);
                int amt = mini-height[i];
                if(amt <0){
                    continue;
                }
                water+=amt;
            }
        }
        return water;
    }
};
