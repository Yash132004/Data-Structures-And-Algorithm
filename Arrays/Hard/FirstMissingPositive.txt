class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        
        // Step 0: Check if 1 exists
        bool hasOne = false;
        for (int x : nums) {
            if (x == 1) {
                hasOne = true;
                break;
            }
        }
        if (!hasOne) return 1;  // 1 is missing
        
        // Step 1: Replace invalid values (<=0 or >n) with 1
        for (int i = 0; i < n; i++) {
            if (nums[i] <= 0 || nums[i] > n) {
                nums[i] = 1;
            }
        }
        
        // Step 2: Mark presence
        for (int i = 0; i < n; i++) {
            int x = abs(nums[i]);
            if (x >= 1 && x <= n) {
                int idx = x - 1;
                if (nums[idx] > 0) {
                    nums[idx] = -nums[idx];
                }
            }
        }
        
        // Step 3: Find the first missing positive
        for (int i = 0; i < n; i++) {
            if (nums[i] > 0) {
                return i + 1;
            }
        }
        
        return n + 1;
    }
};
