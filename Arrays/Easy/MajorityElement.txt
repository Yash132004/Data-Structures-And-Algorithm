class Solution{ //Moore's Voting Algorithm   Algo: 1
    public:
    int majorityElement(vector<int> &nums){
        int candidate; 
        int n = nums.size(), count = 0;
        for(int i =0 ; i<n; i++){
            if(count == 0) candidate = nums[i];
            if(candidate == nums[i]) count++;
            else count--;
        }
        return candidate;
    }
};
