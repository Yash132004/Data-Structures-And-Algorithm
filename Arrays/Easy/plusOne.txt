class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
       int n = digits.size() - 1;
       for(int i = n; i>=0; i--){
        if(digits[i] < 9){
            digits[i]++;  
            return digits;  // <-- exits immediately when increment a digit.
        }else
            digits[i] = 0;
       }
        // if every digit is 9;
       digits.insert(digits.begin(),1);
       return digits;

    }
};
