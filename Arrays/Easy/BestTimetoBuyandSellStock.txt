class Solution {  // One Pass
public:
    int maxProfit(vector<int>& prices) {
        int minPrice = INT_MAX;
        int maxProfit = 0;

        for (int price : prices) {
            minPrice = min(minPrice, price);            // best buy seen so far
            maxProfit = max(maxProfit, price - minPrice); // profit if selling today
        }

        return maxProfit;
    }
};
