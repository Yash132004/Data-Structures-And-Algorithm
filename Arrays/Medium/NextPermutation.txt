class Solution {
public:
    void nextPermutation(vector<int>& nums) {
        int n = nums.size() - 1; 
        int i = n - 1;

        // 1️. Find the first index i from the right where nums[i] < nums[i+1]
        while (i >= 0 && nums[i] >= nums[i + 1]) {
            i--;
        }

        // 2️. If such an index exists, find j (rightmost element "greater than nums[i]")
        if (i >= 0) {
            int j = n;
            while (nums[j] <= nums[i]) { // going right to left because suffix is in dscending order
                j--;
            }
            swap(nums[i], nums[j]);
        }

        // 3️. Reverse the suffix starting at i+1
        reverse(nums.begin() + i + 1, nums.end());
    }
};
