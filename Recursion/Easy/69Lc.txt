class Solution {
public:

    int binarySearch(int n){
        int s = 0, e = n;
        int mid = s + (e-s)/2;
    
        int ans = -1;
        while(s<=e){
            long long square = 1LL*mid*mid;

            if(square == n){
                return mid;
            }
            if(square < n){
                ans = mid;
                s = mid + 1;
            }else{
                e = mid - 1;
            }
            mid = s + (e-s)/2;
        }
        
        return ans;
    }
    int mySqrt(int x) {
        return binarySearch(x);
    }
};
