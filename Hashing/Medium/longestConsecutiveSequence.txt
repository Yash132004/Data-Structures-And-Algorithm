class Solution {
public:
    int longestConsecutive(vector<int>& nums) { 
        int n = nums.size();
        if(n == 0) return 0;
        int longest = 1;
        unordered_set<int> st;
        for(int i = 0; i<n; i++){ //TC: O(n) worst very rare 
            st.insert(nums[i]);
        }
        for(auto it: st){ //TC: O(3n) & SC:O(n).
            if(st.find(it - 1) == st.end()){ // we don't have it-1 element in nums , so the startin point is it.
                int cnt = 1;
                int x = it;
                while(st.find(x+1) != st.end()){ // if we have x+1 means consecutive next element then calculating length.
                    x += 1;
                    cnt += 1;
                }
                longest = max(longest, cnt);
            }
        }
        return longest;
    }
};
