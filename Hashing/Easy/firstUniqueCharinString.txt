class Solution {
public:
    int firstUniqChar(string s) {
        unordered_map<char,int>map;
        for(char ch: s){
            map[ch]++;
        }
        int index = 0;
        for(char ch: s){
            if(map[ch] == 1){
                return index;
            }
            index++;
        }

        return -1;
    }
};
