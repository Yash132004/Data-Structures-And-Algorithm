class Solution {
public:
    int findPeakElement(vector<int>& nums) {
        int start = 0, end = nums.size() - 1;
        int mid ;
        while(start < end){
            mid = start + (end - start)/2;
            if((mid == 0 || nums[mid] > nums[mid-1]) && (mid == nums.size() - 1 || nums[mid] > nums[mid+1])){
                return mid;
            }else if(mid > 0 && nums[mid] < nums[mid - 1]){
                end = mid - 1;
            }else {
                start = mid + 1;
            }
            
        }
        return start;
    }
};
