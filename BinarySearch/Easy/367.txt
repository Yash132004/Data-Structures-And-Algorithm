class Solution {
public:
    bool isPerfectSquare(int num) {
        if (num < 2) return true;  // handles 0 and 1

        int st = 1, end = num / 2;

        while (st <= end) {
            int mid = st + (end - st) / 2;
            long long sq = 1LL * mid * mid;

            if (sq == num) {
                return true;
            } else if (sq > num) {
                end = mid - 1;
            } else {
                st = mid + 1;
            }
        }
        return false;
    }
};
