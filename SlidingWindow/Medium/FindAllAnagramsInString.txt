class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        vector<int>ans;
        vector<int>freq(26,0);
        for(char ch: p){
            freq[ch-'a']++;
        }

        for(int i = 0; i < s.length(); i++){
            int winSize = p.length();
            int k = i;
            vector<int>winFreq(26,0);
            while(k < s.length() && k < i+winSize){
                char ch = s[k++];
                winFreq[ch - 'a']++;
            }
            if(winFreq == freq){
                ans.push_back(i);
            }
        }
        return ans;
    }
};
