class Solution {
public:
    ListNode* rotateRight(ListNode* head, int k) {
        if (!head || k == 0) return head;

        vector<int> arr;
        ListNode* temp = head;

        // store list in array
        while (temp != nullptr) {
            arr.push_back(temp->val);
            temp = temp->next;
        }

        int n = arr.size();
        k = k % n;
        if (k == 0) return head;

        int pos = n - k;

        // rotate using reverse trick
        reverse(arr.begin(), arr.begin() + pos);
        reverse(arr.begin() + pos, arr.end());
        reverse(arr.begin(), arr.end());

        // copy array back to list
        temp = head;
        int j = 0;
        while (temp != nullptr) {
            temp->val = arr[j++];
            temp = temp->next;
        }

        return head;
    }
};
