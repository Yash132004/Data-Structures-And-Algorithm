class Solution {
public:
    ListNode* swapPairs(ListNode* head) {
        ListNode* temp = head;
        int help = 0;

        if (!temp || !temp->next) {
            return head;
        }

        while (temp && temp->next) {
            help = temp->val;
            temp->val = temp->next->val;
            temp->next->val = help;
            
            temp = temp->next->next;  // move ahead by 2
        }

        return head;
    }
};
