class Solution {
public:
    ListNode* oddEvenList(ListNode* head) {
        if (!head) return nullptr;

        ListNode odd(0);
        ListNode* oddl = &odd;

        ListNode even(0);
        ListNode* evenl = &even;

        ListNode* temp = head;
        int pos = 0;

        while (temp) {
            ListNode* nxt = temp->next;  // save next
            temp->next = nullptr;        // break old link

            if ((++pos) % 2 != 0) {      // odd position
                oddl->next = temp;
                oddl = oddl->next;
            } else {                      // even position
                evenl->next = temp;
                evenl = evenl->next;
            }

            temp = nxt;
        }

        oddl->next = even.next;  // join lists
        return odd.next;
    }
};
