/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    bool isPalindrome(ListNode* head) {
        ListNode* slow = head;
        ListNode* fast = head;

        while(fast->next && fast->next->next){
            slow = slow -> next;
            fast = fast->next->next;
        }

        //now reversing the next half
        ListNode* prev = NULL;
        ListNode*curr = slow -> next;
        while(curr){
            ListNode* next = curr -> next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }

        //Comparing the first half with Reverse of second half
        ListNode* left = head;
        ListNode* right = prev;
        while(right){
            if((left->val) != (right->val))
                return false;
            left= left -> next;
            right = right -> next;
        }

        return true;
    }
};
